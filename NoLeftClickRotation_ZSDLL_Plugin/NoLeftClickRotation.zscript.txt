// NoLeftClickRotation - Disable Left-Button View Rotation (ZScript UI + DLL).
//
// Official ZPlugin: compile this file into a .zsc on the target ZBrush,
// then place it into ZStartup\ZPlugs64 (see install notes).
//
// Layout follows the official MiddleButton plugin:
// - the ZScript creates the official UI and detects the canvas state
//   (MouseHPos/VPos + PixolPick material index), pushing it to the DLL;
// - all input interception (IAT hook + window subclass) lives in the DLL.
//
// State pushed to the DLL: 0 = other, 1 = blank canvas, 2 = mesh.

[VarDef, nlcDllPath, ""]
[VarDef, nlcInstalled, 0]
[VarDef, nlcPal, ""]
[VarDef, nlcBody, ""]
[VarDef, nlcSw, "Enable"]
[VarDef, nlcInfo, "Left-drag on blank canvas won't rotate or pan the view; strokes start when the cursor reaches the mesh."]
[VarDef, nlcSwPath, ""]
[VarDef, nlcState, 1]

// loop variables
[VarDef, nlcEdit, 0]
[VarDef, nlcMx, 0]
[VarDef, nlcMy, 0]
[VarDef, nlcDw, 0]
[VarDef, nlcDh, 0]
[VarDef, nlcMat, 0]
[VarDef, nlcZsState, 0]

// locate the DLL: prefer the NoLeftClickRotationData folder next to the .zsc.
[VarSet, nlcDllPath, [FileNameResolvePath, "NoLeftClickRotationData/NoLeftClickRotation.dll"]]
[If, [FileExists, [Var, nlcDllPath]] = 0,
  [VarSet, nlcDllPath, "ZBRUSH_ZSTARTUP/ZPlugs64/NoLeftClickRotationData/NoLeftClickRotation.dll"]
]
[If, [FileExists, [Var, nlcDllPath]] = 0,
  [VarSet, nlcDllPath, "ZBRUSH_ZPLUGS/NoLeftClickRotationData/NoLeftClickRotation.dll"]
]
[If, [FileExists, [Var, nlcDllPath]] = 1,
  [VarSet, nlcInstalled, [FileExecute, [Var, nlcDllPath], "Install"]]
]

[VarSet, nlcPal, "Zplugin:Disable Left-Button View Rotation"]
[VarSet, nlcBody, [StrMerge, nlcPal, ":Body"]]
[VarSet, nlcSwPath, [StrMerge, nlcBody, ":", nlcSw]]
[VarSet, nlcState, [FileExecute, [Var, nlcDllPath], "GetEnabled"]]

// Official UI: outer palette shows the plugin name, inner Body hides its
// title so the switch is not covered.
[ISubPalette, nlcPal, 0]
[ISubPalette, nlcBody, 2]
[ISwitch, nlcSwPath, nlcState, nlcInfo,
  [FileExecute, [Var, nlcDllPath], "SetEnabled", "", 1],
  [FileExecute, [Var, nlcDllPath], "SetEnabled", "", 0],
  0, 0, 0]
[IEnable, nlcSwPath]

// Canvas state detection loop. Event mask 13 = timer | LButton down | LButton
// up (no per-move wakeups). The 50ms timer polls the mouse position and the
// button events update the state right at press/release. The DLL keeps the
// latest state, so if another plugin/macro takes over the single ZScript
// slot, behavior degrades to the last pushed state instead of failing.
[Sleep, 0.05,
  [If, nlcInstalled = 0,
    [If, [FileExists, [Var, nlcDllPath]] = 1,
      [VarSet, nlcInstalled, [FileExecute, [Var, nlcDllPath], "Install"]]
    ]
  ]
  [VarSet, nlcMx, [MouseHPos]]
  [VarSet, nlcMy, [MouseVPos]]
  [VarSet, nlcZsState, 0]
  [If, [FileExists, [Var, nlcDllPath]] = 1,
    [VarSet, nlcEdit, [IGet, Transform:Edit]]
    [If, nlcEdit = 1,
      [VarSet, nlcDw, [IGet, Document:Width]]
      [VarSet, nlcDh, [IGet, Document:Height]]
      [If, (nlcMx>=0)&&(nlcMx<nlcDw)&&(nlcMy>=0)&&(nlcMy<nlcDh),
        [VarSet, nlcMat, [PixolPick, 5, nlcMx, nlcMy]]
        [If, nlcMat = 0, [VarSet, nlcZsState, 1], [VarSet, nlcZsState, 2]],
        [VarSet, nlcZsState, 0]
      ]
    ,
      [VarSet, nlcZsState, 0]
    ]
    [FileExecute, [Var, nlcDllPath], "UpdateState", "", nlcZsState]
  ]
  [SleepAgain, 0.05, 13]
, 13]
