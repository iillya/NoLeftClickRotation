// NoLeftClickRotation - disable left-button view rotation.
// Official ZPlugin entry (compile this text on the ZBrush version you use).
//
// Loads the companion DLL through the official [FileExecute] interface
// (supported on ZBrush 2022+). Creates the official ZScript UI; the label
// follows the Windows system UI language.
//
// - ZBrush 2026+: the DLL uses the embedded Python VM (init.py + official
//   zbrush.commands API) for exact canvas checks, and init.py replaces this
//   UI with the Python UI a moment later.
// - ZBrush 2022-2025: no Python API. The documented ZScript event loop below
//   (Sleep + IGet Transform:Edit + MouseHPos/VPos + PixolPick) pushes the
//   exact canvas state to the DLL on mouse move / left-button down. Because
//   ZBrush only allows one running ZScript at a time, the DLL also keeps an
//   F2 cursor-calibration fallback for when another plugin/macro interrupts
//   the loop.

[VarDef, nlcDllPath, ""]
[VarDef, nlcInstalled, 0]
[VarDef, nlcLang, 1]
[VarDef, nlcPal, ""]
[VarDef, nlcBody, ""]
[VarDef, nlcSw, ""]
[VarDef, nlcInfo, ""]
[VarDef, nlcSwPath, ""]
[VarDef, nlcState, 1]
[VarDef, nlcZsEdit, 0]
[VarDef, nlcZsMx, 0]
[VarDef, nlcZsMy, 0]
[VarDef, nlcZsDw, 0]
[VarDef, nlcZsDh, 0]
[VarDef, nlcZsMat, 0]
[VarDef, nlcZsState, 0]

[VarSet, nlcDllPath, [FileNameResolvePath, "NoLeftClickRotationData/NoLeftClickRotation.dll"]]
[If, [FileExists, [Var, nlcDllPath]] = 0,
  [VarSet, nlcDllPath, "ZBRUSH_ZSTARTUP/ZPlugs64/NoLeftClickRotationData/NoLeftClickRotation.dll"]
]
[If, [FileExists, [Var, nlcDllPath]] = 0,
  [VarSet, nlcDllPath, "ZBRUSH_ZPLUGS/NoLeftClickRotationData/NoLeftClickRotation.dll"]
]
[If, [FileExists, [Var, nlcDllPath]] = 1,
  [VarSet, nlcInstalled, [FileExecute, [Var, nlcDllPath], "Install"]]
  [VarSet, nlcLang, [FileExecute, [Var, nlcDllPath], "Language"]]
]
[If, [Var, nlcLang] = 0,
  [VarSet, nlcPal, "Zplugin:禁用左键视图旋转"]
  [VarSet, nlcSw, "启用"]
  [VarSet, nlcInfo, "左键按住空白画布不再旋转视图；备用校准：空白画布上按 F2。"]
,
  [VarSet, nlcPal, "Zplugin:Disable Left-Button View Rotation"]
  [VarSet, nlcSw, "Enable"]
  [VarSet, nlcInfo, "Left-drag on blank canvas won't rotate the view; fallback: press F2 over blank canvas."]
]
[VarSet, nlcBody, [StrMerge, nlcPal, ":Body"]]
[VarSet, nlcSwPath, [StrMerge, nlcBody, ":", nlcSw]]
[VarSet, nlcState, [FileExecute, [Var, nlcDllPath], "GetEnabled"]]

[ISubPalette, nlcPal, 0]
[ISubPalette, nlcBody, 2]
[ISwitch, nlcSwPath, nlcState, nlcInfo, [FileExecute, [Var, nlcDllPath], "SetEnabled", "", 1], [FileExecute, [Var, nlcDllPath], "SetEnabled", "", 0], 0, 0, 0]
[IEnable, nlcSwPath]

// Legacy state feed: exact blank/mesh judgment for ZBrush 2022-2025.
// Timer wake (event 1) + bare SleepAgain (documented conservative form).
[Sleep, 0.05,
  [VarSet, nlcZsState, 0]
  [If, [FileExists, [Var, nlcDllPath]] = 1,
    [VarSet, nlcZsEdit, [IGet, Transform:Edit]]
    [If, nlcZsEdit = 1,
      [VarSet, nlcZsMx, [MouseHPos]]
      [VarSet, nlcZsMy, [MouseVPos]]
      [VarSet, nlcZsDw, [IGet, Document:Width]]
      [VarSet, nlcZsDh, [IGet, Document:Height]]
      [If, (nlcZsMx>=0)&&(nlcZsMx<nlcZsDw)&&(nlcZsMy>=0)&&(nlcZsMy<nlcZsDh),
        [VarSet, nlcZsMat, [PixolPick, 5, nlcZsMx, nlcZsMy]]
        [If, nlcZsMat = 0, [VarSet, nlcZsState, 1], [VarSet, nlcZsState, 2]],
        [VarSet, nlcZsState, 0]
      ]
    ,
      [VarSet, nlcZsState, 0]
    ]
    [FileExecute, [Var, nlcDllPath], "UpdateState", "", nlcZsState]
  ]
  [SleepAgain]
, 1]
