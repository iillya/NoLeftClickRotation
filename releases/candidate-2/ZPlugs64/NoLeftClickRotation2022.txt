// No Left Click Rotation - Candidate 2 (RC2)
// ZBrush 2022+ on-demand PixolPick bridge

[VarDef,nlrDll,""]
[VarSet,nlrDll,[FileNameResolvePath,"NoLeftClickRotation2022Data/NoLeftClickRotation2022.dll"]]
[VarDef,nlrEnabled,1]
[VarDef,nlrDelay,0]
[VarDef,nlrWindow,-1]
[VarDef,nlrMat,0]
[VarDef,nlrStateBits,0]
[VarDef,nlrEditValue,0]
[VarDef,nlrEditFlags,0]
[VarDef,nlrEditDetected,0]
[VarDef,nlrBridgeHotkey,0]
[VarDef,nlrBridgeHotkeyBefore,0]
[VarDef,nlrLock,0]

// This routine is called only through the private hotkey. RC2 synchronizes
// left-button classification state and returns one PixolPick material sample.
[RoutineDef,NLRBridge,
    // Detect and repair a runtime binding cleared by another hotkey load.
    [VarSet,nlrBridgeHotkeyBefore,[IGetHotkey,"ZPlugin:No Left Click Rotation:Pixol Bridge"]]
    [If,[Var,nlrBridgeHotkeyBefore] != 91,
        [ISetHotkey,"ZPlugin:No Left Click Rotation:Pixol Bridge",91]
    ]
    [VarSet,nlrStateBits,[Var,nlrEnabled]]
    [VarSet,nlrEditValue,[IGet,Transform:Edit]]
    [VarSet,nlrEditFlags,[IGetFlags,Transform:Edit]]
    [VarSet,nlrEditDetected,0]
    [If,[Var,nlrEditValue] == 1,
        [VarSet,nlrEditDetected,1]
    ]
    // Official ZScript guidance: Edit is pressed when its flags are >= 8.
    [If,[Var,nlrEditFlags] >= 8,
        [VarSet,nlrEditDetected,1]
    ]
    [If,[Var,nlrEditDetected],
        [VarAdd,nlrStateBits,2]
    ]
    [VarSet,nlrWindow,[IGet,"Preferences:Utilities:View Window Id"]]
    [VarSet,nlrBridgeHotkey,[IGetHotkey,"ZPlugin:No Left Click Rotation:Pixol Bridge"]]

    [FileExecute,[Var,nlrDll],NlrSync,[StrMerge,"",[Var,nlrWindow],",",[Var,nlrEditValue],",",[Var,nlrEditFlags],",",[Var,nlrBridgeHotkeyBefore],",",[Var,nlrBridgeHotkey]],[Var,nlrStateBits],nlrMem,nlrMem]

    // Python V1 camera rule: enabled + Edit + right button up = locked.
    [MemRead,nlrMem,nlrLock,0,0]
    [If,[Var,nlrLock] == 1,
        [ISet,"Draw:Lock Camera",1]
    ]
    [If,[Var,nlrLock] == 0,
        [ISet,"Draw:Lock Camera",0]
    ]
    // Right DOWN is replayed only after the unlock has reached ZBrush.
    [FileExecute,[Var,nlrDll],ReplayRightDown]

    [VarSet,nlrMat,[PixolPick,5,[MouseHPos],[MouseVPos]]]
    [FileExecute,[Var,nlrDll],HotkeyPixol,,[Var,nlrMat]]
]

[ISubPalette,"ZPlugin:No Left Click Rotation"]

[ISwitch,"ZPlugin:No Left Click Rotation:Enable",1,
    "Enable the ZBrush compatibility bridge.",
    [VarSet,nlrEnabled,1]
    [FileExecute,[Var,nlrDll],SetEnabled,,1]
,
    [VarSet,nlrEnabled,0]
    [FileExecute,[Var,nlrDll],SetEnabled,,0]
    [ISet,"Draw:Lock Camera",0]
,0,1,0]

[ISlider,"ZPlugin:No Left Click Rotation:Sculpt Start Delay",0,1,0,10,
    "Delay in milliseconds before the bridged sculpt stroke starts.",
    [VarSet,nlrDelay,[IGet,"ZPlugin:No Left Click Rotation:Sculpt Start Delay"]]
    [FileExecute,[Var,nlrDll],SetDelay,,[Var,nlrDelay]]
,0,1,0]

// Internal on-demand bridge. F12 uses ZBrush's unmodified key code 91.
[IButton,"ZPlugin:No Left Click Rotation:Pixol Bridge",
    "Internal on-demand PixolPick bridge.",
    [RoutineCall,NLRBridge]
,0,1,0]
[ISetHotkey,"ZPlugin:No Left Click Rotation:Pixol Bridge",91]

[FileExecute,[Var,nlrDll],Init]
[If,[FileExists,[FileNameResolvePath,"NoLeftClickRotation2022Data/NoLeftClickRotation2022.dll"]],
    [If,[MemGetSize,nlrMem] == 0,
        [MemCreate,nlrMem,8,0]
    ]
    // The switch is created pressed. IGet may briefly report zero while the
    // startup script is still registering its new interface items.
    [VarSet,nlrEnabled,1]
    [VarSet,nlrDelay,[IGet,"ZPlugin:No Left Click Rotation:Sculpt Start Delay"]]
    [FileExecute,[Var,nlrDll],SetEnabled,,[Var,nlrEnabled]]
    [FileExecute,[Var,nlrDll],SetDelay,,[Var,nlrDelay]]
    [ISetHotkey,"ZPlugin:No Left Click Rotation:Pixol Bridge",91]
,
    [MessageOK,"NoLeftClickRotation2022.dll not found in ZPlugs64."]
]

[IEnable,"ZPlugin:No Left Click Rotation:Enable"]
[IEnable,"ZPlugin:No Left Click Rotation:Sculpt Start Delay"]
[IEnable,"ZPlugin:No Left Click Rotation:Pixol Bridge"]
