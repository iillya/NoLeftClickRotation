// No Left Click Rotation - ZBrush 2022 bridge

[VarDef,nlrDll,""]
[VarSet,nlrDll,[FileNameResolvePath,"NoLeftClickRotation2022Data/NoLeftClickRotation2022.dll"]]
[VarDef,nlrEnabled,1]
[VarDef,nlrDelay,0]
[VarDef,nlrEvent,0]
[VarDef,nlrWindow,-1]
[VarDef,nlrMat,0]
[VarDef,nlrNeed,0]
[VarDef,nlrStateBits,0]
[VarDef,nlrLock,-1]
[VarDef,nlrLockPrev,-1]
[VarDef,nlrLockTick,0]
[VarDef,nlrHoverNeed,0]
[VarDef,nlrRight,0]
[VarDef,nlrTraceTick,0]

[RoutineDef,NLRWake,
    // Temporary diagnostic: log the wake event code every 50 wakes.
    [VarAdd,nlrTraceTick,1]
    [If,[Var,nlrTraceTick] >= 50,
        [VarSet,nlrTraceTick,0]
        [FileExecute,[Var,nlrDll],Trace,[StrMerge,"ev=",[Var,nlrEvent]]]
    ]
    [If,[Var,nlrEvent] == 512,
        // Another ZScript/plugin took focus (ZBrush allows only one sleeping
        // script at a time). Release the camera and re-arm the sleep waiting
        // for the focus/startup event so the loop resumes when focus returns.
        // The DLL must stay alive (no Shutdown here) so the poke timer keeps
        // running and can wake us again.
        [FileExecute,[Var,nlrDll],Trace,"ev=512 focus lost"]
        [ISet,"Draw:Lock Camera",0]
        [SleepAgain,0.005,1+2+4+8+64+128+256+512]
    ,
        // Re-sync from the actual switch every wake: at startup the switch is
        // created pressed (display shows enabled) but IGet inside the same
        // script run can return 0 before ZBrush registers the item, which
        // left the plugin inactive until the user toggled it by hand.
        [VarSet,nlrEnabled,[IGet,"ZPlugin:No Left Click Rotation:Enable"]]
        [VarSet,nlrStateBits,[Var,nlrEnabled]]
        [If,[IGet,Transform:Edit],
            [VarAdd,nlrStateBits,2]
        ]

        // Window ID source matches Python V1 (Preferences debug readback).
        [VarSet,nlrWindow,[IGet,"Preferences:Utilities:View Window Id"]]

        // One reliable round-trip: the DLL writes lock/need flags into the
        // memory block (FileExecute return values are unreliable in 2026).
        // Per the plugin API the block pointer arrives in the "input" slot;
        // the DLL writes there and ZScript reads it back.
        [FileExecute,[Var,nlrDll],NlrSync,[StrMerge,"",[Var,nlrWindow]],[Var,nlrStateBits],nlrMem,nlrMem]
        [MemRead,nlrMem,nlrLock,0,0]
        [MemRead,nlrMem,nlrNeed,0,4]
        [MemRead,nlrMem,nlrHoverNeed,0,8]
        [MemRead,nlrMem,nlrRight,0,12]

        // Camera lock is a switch: press = locked, unpress = unlocked.
        // Sync the desired state directly (change + periodic re-assert like
        // Python V1) instead of gating on the button readback.
        [If,[Var,nlrLock],
            // Right button released: finish any queued release first, then
            // keep the camera locked (periodic re-assert like Python V1).
            [FileExecute,[Var,nlrDll],ReplayRightDown]
            [If,[Var,nlrLock] != [Var,nlrLockPrev],
                [ISet,"Draw:Lock Camera",1]
                [VarSet,nlrLockPrev,1]
                [VarSet,nlrLockTick,0]
            ,
                [VarAdd,nlrLockTick,1]
                [If,[Var,nlrLockTick] >= 40,
                    [ISet,"Draw:Lock Camera",1]
                    [VarSet,nlrLockTick,0]
                ]
            ]
        ,
            // Right button held: unlock the camera first, then replay the
            // swallowed press so ZBrush starts the rotate gesture.
            [If,[Var,nlrLock] != [Var,nlrLockPrev],
                [ISet,"Draw:Lock Camera",0]
                [VarSet,nlrLockPrev,0]
                [VarSet,nlrLockTick,0]
            ,
                [VarAdd,nlrLockTick,1]
                [If,[Var,nlrLockTick] >= 40,
                    [ISet,"Draw:Lock Camera",0]
                    [VarSet,nlrLockTick,0]
                ]
            ]
            [FileExecute,[Var,nlrDll],ReplayRightDown]
        ]

        [If,[Var,nlrNeed],
            [VarSet,nlrMat,[PixolPick,5,[MouseHPos],[MouseVPos]]]
            [FileExecute,[Var,nlrDll],UpdatePixol,,[Var,nlrMat]]
        ]

        // Hover cache: sample the material while idle so a model press can be
        // passed through immediately without the probe delay. The DLL gates
        // this to the canvas and throttles it to avoid PixolPick overload.
        [If,[Var,nlrHoverNeed],
            [VarSet,nlrMat,[PixolPick,5,[MouseHPos],[MouseVPos]]]
            [FileExecute,[Var,nlrDll],SetHoverMat,,[Var,nlrMat]]
        ]

        [SleepAgain,0.005,1+2+4+8+64+128+512]
    ]
]

[RoutineDef,NLRStartSleep,
    // Mask includes 256 (Startup): when ZBrush auto-loads this plugin at
    // startup, timer wakes may not be dispatched until startup finishes, so
    // the Startup event is what starts the loop. The timer (1) covers manual
    // reloads after ZBrush is already running.
    [Sleep,0.005,
        [RoutineCall,NLRWake],
        1+2+4+8+64+128+256+512,
        nlrEvent,
        nlrWindow
    ]
]

[ISubPalette,"ZPlugin:No Left Click Rotation"]

[ISwitch,"ZPlugin:No Left Click Rotation:Enable",1,
    "Enable the ZBrush 2022 compatibility bridge.",
    [VarSet,nlrEnabled,1]
    [FileExecute,[Var,nlrDll],SetEnabled,,1]
,
    [VarSet,nlrEnabled,0]
    [FileExecute,[Var,nlrDll],SetEnabled,,0]
    [ISet,"Draw:Lock Camera",0]
    [VarSet,nlrLockPrev,0]
    [VarSet,nlrLockTick,0]
,0,1,0]

[ISlider,"ZPlugin:No Left Click Rotation:Sculpt Start Delay",0,1,0,10,
    "Delay in milliseconds before the bridged sculpt stroke starts.",
    [VarSet,nlrDelay,[IGet,"ZPlugin:No Left Click Rotation:Sculpt Start Delay"]]
    [FileExecute,[Var,nlrDll],SetDelay,,[Var,nlrDelay]]
,0,1,0]

// Initialize automatically when ZBrush loads the plugin. Note: ZBrush 2026
// does not reliably return the DLL's result from FileExecute, so the return
// value of Init cannot be used for control flow. The DLL logs its own
// initialization steps; here we only need the DLL file to be present.
[FileExecute,[Var,nlrDll],Init]
[FileExecute,[Var,nlrDll],Trace,"init: after Init"]
[If,[FileExists,[FileNameResolvePath,"NoLeftClickRotation2022Data/NoLeftClickRotation2022.dll"]],
    [FileExecute,[Var,nlrDll],Trace,"init: file ok"]
    [If,[MemGetSize,nlrMem] == 0,
        [MemCreate,nlrMem,16,0]
    ]
    [FileExecute,[Var,nlrDll],Trace,"init: mem ok"]
    [VarSet,nlrEnabled,[IGet,"ZPlugin:No Left Click Rotation:Enable"]]
    [FileExecute,[Var,nlrDll],Trace,[StrMerge,"init: en=",[Var,nlrEnabled]]]
    [VarSet,nlrDelay,[IGet,"ZPlugin:No Left Click Rotation:Sculpt Start Delay"]]
    [FileExecute,[Var,nlrDll],SetEnabled,,[Var,nlrEnabled]]
    [FileExecute,[Var,nlrDll],SetDelay,,[Var,nlrDelay]]
    [FileExecute,[Var,nlrDll],Trace,"init: starting sleep"]
    [RoutineCall,NLRStartSleep]
,
    [FileExecute,[Var,nlrDll],Trace,"init: FILE MISSING"]
    [MessageOK,"NoLeftClickRotation2022.dll not found in ZPlugs64."]
]

// Keep ZScript-generated interface items enabled when other plugins load.
[IEnable,"ZPlugin:No Left Click Rotation:Enable"]
[IEnable,"ZPlugin:No Left Click Rotation:Sculpt Start Delay"]
